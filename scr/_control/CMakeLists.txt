cmake_minimum_required(VERSION 3.21)
project(SixMagControl VERSION 1.0 LANGUAGES CXX)
add_library(SixMagControl STATIC MagneticModel.cpp MagneticModel.h)
add_library(SixMag::Control ALIAS SixMagControl)
target_compile_features(SixMagControl PUBLIC cxx_std_17)
target_include_directories(SixMagControl PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
if(MSVC)
    target_compile_options(SixMagControl PRIVATE /W4 /fp:precise)
else()
    target_compile_options(SixMagControl PRIVATE -Wall -Wextra -Wpedantic
        -fno-math-errno -ffp-contract=off)
endif()
# No fast-math, CPU-specific ISA, Qt, MATLAB or third-party math runtime.
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
    set(CONTROL_TESTS_DEFAULT ON)
else()
    set(CONTROL_TESTS_DEFAULT OFF)
endif()
option(SIXMAG_CONTROL_BUILD_TESTS "Build tests and benchmark" ${CONTROL_TESTS_DEFAULT})
if(SIXMAG_CONTROL_BUILD_TESTS)
    enable_testing()
    find_package(Threads REQUIRED)
    add_executable(SixMagControlTests tests/NativeModelTests.cpp)
    target_link_libraries(SixMagControlTests PRIVATE SixMag::Control Threads::Threads)
    add_test(NAME MagneticModelReference COMMAND SixMagControlTests
        "${CMAKE_CURRENT_SOURCE_DIR}/tests/reference_cases.csv")
    add_executable(SixMagControlBenchmark tests/NativeModelBenchmark.cpp)
    target_link_libraries(SixMagControlBenchmark PRIVATE SixMag::Control)
endif()
